# Files in client/public/ are copied verbatim into client/dist by Vite, so this
# is served as a real text/plain file. Vercel checks the filesystem before it
# applies the SPA rewrite in vercel.json, which is why this must exist as a file:
# without it, /robots.txt fell through the rewrite and returned index.html, and
# crawlers read HTML where they expected directives.

User-agent: *
Allow: /

# Allow crawlers to observe the initial noindex directive on workspace, share,
# token and unknown routes. A robots disallow would hide that directive and can
# leave externally linked URLs indexed without a snippet. Authentication and
# share tokens enforce access; noindex is only the search policy. Private URLs
# are excluded from the sitemap, and their shells use nofollow.
# Rendering assets also remain crawlable for JavaScript search rendering.
